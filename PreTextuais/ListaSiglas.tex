\begin{siglas}
  \item[5W2H] Ferramenta de planejamento baseada nas perguntas \textit{What, Why, Where, When, Who, How} e \textit{How much}
  \item[6M] Categorias de causas do Diagrama de Ishikawa: mão de obra, método, máquina, material, medição e meio ambiente
  \item[8D] Oito disciplinas para a solução de problemas (\textit{Eight Disciplines of Problem Solving})
  \item[API] Interface de programação de aplicações (\textit{Application Programming Interface})
  \item[GND] Terra do circuito (\textit{ground})
  \item[GPIO] Entrada e saída de uso geral (\textit{General Purpose Input/Output})
  \item[HTTP] Protocolo de transferência de hipertexto (\textit{Hypertext Transfer Protocol})
  \item[IEC] Comissão Eletrotécnica Internacional (\textit{International Electrotechnical Commission})
  \item[IFMG] Instituto Federal de Educação, Ciência e Tecnologia de Minas Gerais
  \item[ISA] Sociedade Internacional de Automação (\textit{International Society of Automation})
  \item[LGPD] Lei Geral de Proteção de Dados Pessoais
  \item[MQTT] Protocolo de mensagens do tipo publicação e assinatura (\textit{Message Queuing Telemetry Transport})
  \item[NA] Contato normalmente aberto
  \item[OBA] Auditoria do produto acabado (\textit{Out-of-Box Audit})
  \item[OData] Protocolo aberto de dados (\textit{Open Data Protocol})
  \item[PDCA] Ciclo planejar, executar, verificar e agir (\textit{Plan, Do, Check, Act})
  \item[PFMEA] Análise de modos de falha e efeitos do processo (\textit{Process Failure Mode and Effects Analysis})
  \item[QRQC] Resposta Rápida para o Controle da Qualidade (\textit{Quick Response to Quality Control})
  \item[REST] Transferência de estado representacional (\textit{Representational State Transfer})
  \item[SQL] Linguagem de consulta estruturada (\textit{Structured Query Language})
  \item[TD] Transformação Digital
  \item[VCC] Alimentação positiva do circuito
\end{siglas}
