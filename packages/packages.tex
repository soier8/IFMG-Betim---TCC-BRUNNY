% Figura de estudo de caso com substituição automática.
% Ao adicionar o arquivo indicado na pasta Figuras, o quadro de orientação
% desaparece e a imagem passa a ser exibida com a mesma legenda e rótulo.
% Uso: \figuraCaso[largura]{arquivo}{legenda}{rotulo}{orientacao}{fonte}
\newcommand{\figuraCaso}[6][0.92\textwidth]{%
    \begin{figure}[htb]
        \centering
        \caption{#3}
        \label{#4}
        \IfFileExists{Figuras/#2}{%
            \includegraphics[width=#1,height=0.72\textheight,keepaspectratio]{Figuras/#2}%
        }{%
            \setlength{\fboxsep}{10pt}%
            \fcolorbox{placeholdergray}{gray!6}{%
                \begin{minipage}[c][5.2cm][c]{0.84\textwidth}
                    \centering
                    \color{placeholdergray}
                    \textbf{ESPAÇO RESERVADO PARA IMAGEM}\\[0.5em]
                    \texttt{Figuras/#2}\\[0.8em]
                    \small #5
                \end{minipage}%
            }%
        }
        \legend{Fonte: #6}
    \end{figure}
    \FloatBarrier
}
